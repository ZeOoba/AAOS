+"""
+tests/_tool_loader.py — helper to import the hyphenated tools/*.py CLI
+scripts in-process for unit testing.
+
+Filenames like "create-task.py" cannot be imported with a plain `import`
+statement (Python identifiers can't contain hyphens), so this module uses
+importlib.util.spec_from_file_location per docs/PHASE3-DESIGN.md's
+suggestion. Loaded modules are cached in sys.modules under a synthetic,
+collision-free name so repeated calls for the same file return the same
+module object (mirroring normal import caching).
+
+tools/*.py scripts do `import _aaos_common as aaos`; this module ensures
+tools/ is on sys.path before exec'ing any of them, the same sys.path
+pattern tests/test_validate.py and tests/test_aaos_common.py already use.
+"""
+from __future__ import annotations
+
+import importlib.util
+import sys
+from pathlib import Path
+
+REPO_ROOT = Path(__file__).resolve().parent.parent
+TOOLS_DIR = REPO_ROOT / "tools"
+
+if str(TOOLS_DIR) not in sys.path:
+    sys.path.insert(0, str(TOOLS_DIR))
+
+
+def load_tool(filename: str):
+    """Load tools/<filename> (e.g. "create-task.py") as a module object.
+
+    Cached per filename in sys.modules so multiple test classes loading the
+    same tool don't re-exec it (and so `module.main` is a stable object
+    across a test run, matching how a normal `import` would behave).
+    """
+    module_name = "_aaos_tool_" + filename.replace("-", "_").replace(".py", "")
+    if module_name in sys.modules:
+        return sys.modules[module_name]
+    path = TOOLS_DIR / filename
+    spec = importlib.util.spec_from_file_location(module_name, path)
+    module = importlib.util.module_from_spec(spec)
+    sys.modules[module_name] = module
+    spec.loader.exec_module(module)
+    return module

+        for i in range(10):
+            task_id = f"t-race-{i}"

+            barrier = threading.Barrier(2)

+            def worker(holder, session):
+                barrier.wait()
+                r = aaos.acquire_lease(
+                    task_id, PROJECT, holder, session,
+                    now=NOW, settle_seconds=0, root=str(self.root),

+                with outcomes_lock:

+            t1 = threading.Thread(target=worker, args=("claude-code", "sess-a"))
+            t2 = threading.Thread(target=worker, args=("cowork", "sess-b"))
+            t1.start()
+            t2.start()
+            t1.join()
+            t2.join()
+
+            self.assertEqual(
+                sorted(o.value for o in outcomes),
+                sorted([aaos.LeaseOutcome.ACQUIRED.value, aaos.LeaseOutcome.DENIED.value]),
+                f"iteration {i}: outcomes were {outcomes!r}",

+def _load_hyphenated_module(name: str, filename: str):
+    path = TOOLS_DIR / filename
+    spec = importlib.util.spec_from_file_location(name, path)
+    module = importlib.util.module_from_spec(spec)
+    spec.loader.exec_module(module)
+    return module

+    """Invoke project-notion.py's main() in-process, capturing stdout.
+    Returns (exit_code, stdout_text)."""
+    buf = io.StringIO()

+def match_write_scope(candidate_path: str, write_scope) -> bool:
+    """True if candidate_path is covered by at least one glob in write_scope.
+
+    Uses fnmatch.fnmatch (not pathlib.PurePath.match) because write_scope
+    globs use "**" to mean "this directory and everything under it,
+    recursively" (see agent-registry.json, e.g. "C:/Users/Owner/AAOS/**").
+    fnmatch treats "*" (and so "**") as matching any run of characters
+    including path separators, which is exactly that recursive semantics;
+    PurePath.match treats "**" as an ordinary single-segment "*" and would
+    require multiple pattern segments to express the same thing.
+
+    Before matching, candidate_path is canonicalized: normalized to forward
+    slashes (this project always authors write_scope globs with forward
+    slashes even for Windows paths) and then run through posixpath.normpath
+    to lexically resolve "." / ".." segments and collapse redundant
+    separators. This is load-bearing, not cosmetic: fnmatch.fnmatch is a
+    pure string match with no notion of path semantics, so without
+    canonicalizing first, a traversal path like
+    "C:/Users/Owner/AAOS/../../Windows/System32/evil.dll" would satisfy the
+    glob "C:/Users/Owner/AAOS/**" merely because it *starts with* that
+    literal prefix, even though it resolves well outside the scoped
+    directory. posixpath (rather than os.path) is used deliberately so this
+    resolution is platform-independent — write_scope globs are an internal,
+    always-forward-slash convention, not literal OS paths, and normpath's
+    ".."-collapsing is purely lexical either way (no filesystem/symlink
+    resolution occurs, nor is any needed here).
+
+    Lifted verbatim from tools/validate.py per docs/PHASE3-DESIGN.md
+    section 1.1 — this is now the single implementation; validate.py
+    imports it from here.
+    """
+    norm_candidate = posixpath.normpath(candidate_path.replace("\\", "/"))
+    for pattern in write_scope:
+        if not isinstance(pattern, str):
+            continue
+        norm_pattern = pattern.replace("\\", "/")
+        if fnmatch.fnmatch(norm_candidate, norm_pattern):
+            return True
+    return False

+def atomic_write(path: Path | str, content: str | bytes) -> None:
+    """The ADR-0002 idiom, verbatim: create parent dirs; unique temp name
+    path + '.tmp-' + str(os.getpid()); write bytes (UTF-8 if str); flush() +
+    os.fsync(); os.replace(tmp, path); on any exception, best-effort unlink
+    the temp and re-raise. Never writes `path` in place."""
+    path = Path(path)
+    os.makedirs(path.parent, exist_ok=True)
+    tmp = Path(str(path) + ".tmp-" + str(os.getpid()))
+    try:
+        data = content.encode("utf-8") if isinstance(content, str) else content
+        with open(tmp, "wb") as fh:
+            fh.write(data)
+            fh.flush()
+            os.fsync(fh.fileno())
+        os.replace(str(tmp), str(path))
+    except Exception:
+        try:
+            if tmp.exists():
+                tmp.unlink()
+        except OSError:
+            pass
+        raise

+def _parse_iso_dt(value) -> datetime:
+    """Parse an ISO-8601 datetime string. Fail-closed CorruptStateError on
+    missing/non-string/unparseable/naive values — never silently assumed
+    UTC (design section 0.3)."""
+    if not isinstance(value, str) or not value:
+        raise CorruptStateError(f"missing or non-string datetime value: {value!r}")
+    try:
+        dt = datetime.fromisoformat(value)
+    except ValueError as exc:
+        raise CorruptStateError(f"unparseable datetime value {value!r}: {exc}") from exc
+    if dt.tzinfo is None:
+        raise CorruptStateError(f"naive datetime not allowed (ambiguous timezone): {value!r}")
+    return dt

+# ---------------------------------------------------------------------------
+# 1.9 Stale-lock reclamation (defined ahead of acquire_lease, which uses it)
+# ---------------------------------------------------------------------------
+
+def reclaim_decision(
+    lock: dict, heartbeat_status: str | None, *, now: datetime | None = None, skew_minutes: int = 5
+) -> ReclaimDecision:
+    resolved_now = _now(now)
+    expires_raw = lock.get("expires") if isinstance(lock, dict) else None
+    try:
+        expires = _parse_iso_dt(expires_raw)
+    except CorruptStateError:
+        return ReclaimDecision(
+            reclaimable=False,
+            classification="",
+            reason="lock has a missing or unparseable 'expires'; fail-closed, not reclaimable",
+        )
+
+    grace_deadline = expires + timedelta(minutes=skew_minutes)
+    if resolved_now <= grace_deadline:
+        return ReclaimDecision(
+            reclaimable=False,
+            classification="",
+            reason=f"lease live: now <= expires+{skew_minutes}m grace ({grace_deadline.isoformat()})",
+        )
+
+    if heartbeat_status in ("missed", "unknown", None):
+        classification = "DeadAgent"
+    else:
+        classification = "ExpiredLapsed"
+    return ReclaimDecision(
+        reclaimable=True,
+        classification=classification,
+        reason=(
+            f"lease expired past grace (now > expires+{skew_minutes}m); "
+            f"heartbeat_status={heartbeat_status!r}"
+        ),
+    )

+def archive_lock(lock_file: Path | str, *, now: datetime | None = None) -> Path:
+    """os.replace to locks/_archive/<task_id>.<compact-ts>.lock; never deletes."""
+    lock_file = Path(lock_file)
+    resolved_now = _now(now)
+    base = lock_file.name.split(".lock", 1)[0]
+    compact_ts = resolved_now.strftime("%Y%m%dT%H%M%S%f")
+    archive_dir = lock_file.parent / "_archive"
+    os.makedirs(archive_dir, exist_ok=True)
+    target = archive_dir / f"{base}.{compact_ts}.lock"
+    if target.exists():
+        target = archive_dir / f"{base}.{compact_ts}.{os.getpid()}.lock"

+# ---------------------------------------------------------------------------
+# 1.7 Lease acquisition — ADR-0003's 6-step protocol

+_CONFLICT_SUFFIX_RE = re.compile(r"^(-[A-Za-z0-9_.-]+)?( \(\d+\))?\.lock$")

+
+def _matches_conflict_shape(filename: str, task_id: str) -> bool:
+    """True for the canonical <task_id>.lock as well as OneDrive
+    conflict-copy shapes: <id>-<HOST>.lock, <id> (N).lock, <id>-<HOST> (N).lock."""
+    if not filename.startswith(task_id):
+        return False
+    suffix = filename[len(task_id):]
+    return bool(_CONFLICT_SUFFIX_RE.match(suffix))
+
+
+def _scan_for_conflicts(locks_dir: Path, task_id: str, holder: str, session: str) -> list[str]:
+    conflicts: list[str] = []
+    if not locks_dir.is_dir():
+        return conflicts
+    canonical_name = f"{task_id}.lock"
+    for entry in sorted(locks_dir.iterdir(), key=lambda p: p.name):
+        if not entry.is_file():
+            continue
+        name = entry.name
+        if not _matches_conflict_shape(name, task_id):
+            continue
+        if name == canonical_name:
+            try:
+                data = load_json(entry)
+            except AaosError:
+                conflicts.append(str(entry))
+                continue
+            if data.get("holder") != holder or data.get("session") != session:
+                conflicts.append(str(entry))
+            continue
+        # Any other matching filename is itself a conflict-copy shape.
+        conflicts.append(str(entry))
+    return conflicts
+

+def _exclusive_create(path: Path, data: bytes) -> None:
+    flags = os.O_CREAT | os.O_EXCL | os.O_WRONLY
+    if hasattr(os, "O_BINARY"):
+        flags |= os.O_BINARY
+    fd = os.open(str(path), flags)
+    try:
+        os.write(fd, data)
+    finally:
+        os.close(fd)

+def acquire_lease(
+    task_id: str, project_id: str, holder: str, session: str, *,
+    now: datetime | None = None,
+    ttl_minutes: int = 45, skew_minutes: int = 5, settle_seconds: float = 30.0,
+    sleep: Callable[[float], None] = time.sleep,
+    reclaim_expired: bool = True,
+    root: str | None = None,
+) -> LeaseResult:
+    resolved_now = _now(now)
+
+    # ---- 1. Precheck ----
+    task = load_task(task_id, project_id, root=root)
+    state = task.get("state")
+    lease = task.get("lease")
+    reclaimed_prior = None
+
+    if state == "queued":
+        pass  # proceed
+    else:
+        expired = False
+        if lease is not None:
+            try:
+                lease_expires = _parse_iso_dt(lease.get("expires"))
+                expired = resolved_now > lease_expires + timedelta(minutes=skew_minutes)
+            except CorruptStateError:
+                expired = False  # fail-closed: an unparseable lease is treated as live, not reclaimable
+        if lease is not None and expired:
+            if reclaim_expired:
+                incumbent_file = lock_path(task_id, project_id, root=root)
+                if incumbent_file.exists():
+                    archive_lock(incumbent_file, now=resolved_now)
+                reclaimed_prior = lease
+                # proceed
+            else:
+                return LeaseResult(
+                    outcome=LeaseOutcome.DENIED,
+                    reason="expired lease present; run expire-leases",
+                )
+        else:
+            return LeaseResult(
+                outcome=LeaseOutcome.DENIED,
+                reason=f"task state={state}, lease live",
+            )
+
+    # ---- 2. Stage ----
+    final_lock_path = lock_path(task_id, project_id, root=root)
+    acquired_iso = resolved_now.isoformat()
+    expires_dt = resolved_now + timedelta(minutes=ttl_minutes)
+    lock_dict = {
+        "protocol_version": "1.0",
+        "task_ref": task_id,
+        "holder": holder,
+        "acquired": acquired_iso,
+        "expires": expires_dt.isoformat(),
+        "session": session,
+    }
+    content = json.dumps(lock_dict, indent=2, ensure_ascii=False)
+    content_bytes = content.encode("utf-8")
+
+    pid = os.getpid()
+    sidecar = final_lock_path.parent / f"{final_lock_path.name}.tmp-{holder}-{pid}"
+    atomic_write(sidecar, content)
+
+    def _cleanup_sidecar():
+        try:
+            if sidecar.exists():
+                sidecar.unlink()
+        except OSError:
+            pass
+
+    # ---- 3. Exclusive create ----
+    os.makedirs(final_lock_path.parent, exist_ok=True)
+    try:
+        _exclusive_create(final_lock_path, content_bytes)
+        _cleanup_sidecar()
+    except FileExistsError:
+        try:
+            incumbent = load_lock(task_id, project_id, root=root)
+        except CorruptStateError:
+            _cleanup_sidecar()
+            raise
+
+        incumbent_live = True
+        if incumbent is not None:
+            try:
+                incumbent_expires = _parse_iso_dt(incumbent.get("expires"))
+                incumbent_live = resolved_now <= incumbent_expires + timedelta(minutes=skew_minutes)
+            except CorruptStateError:
+                incumbent_live = True  # fail-closed: don't steal an ambiguous incumbent
+
+        if not incumbent_live:
+            archive_lock(final_lock_path, now=resolved_now)
+            try:
+                _exclusive_create(final_lock_path, content_bytes)
+                _cleanup_sidecar()
+            except FileExistsError:
+                archive_lock(sidecar, now=resolved_now)
+                return LeaseResult(
+                    outcome=LeaseOutcome.CONTESTED,
+                    conflict_paths=[str(final_lock_path)],
+                    reason="collision persisted after reclaim-and-retry",
+                )
+        else:
+            _cleanup_sidecar()
+            return LeaseResult(outcome=LeaseOutcome.DENIED, reason="incumbent lease live")
+
+    # ---- 4. Settle ----
+    sleep(settle_seconds)
+
+    # ---- 5. Rescan ----
+    conflict_paths = _scan_for_conflicts(final_lock_path.parent, task_id, holder, session)
+    if conflict_paths:
+        if final_lock_path.exists():
+            archive_lock(final_lock_path, now=resolved_now)
+        return LeaseResult(
+            outcome=LeaseOutcome.CONTESTED,
+            conflict_paths=conflict_paths,
+            reason="conflict lock(s) found during settle rescan",
+        )
+
+    # ---- 6. Confirm ----
+    return LeaseResult(
+        outcome=LeaseOutcome.ACQUIRED,
+        lock=lock_dict,
+        expires=expires_dt,
+        reclaimed_prior=reclaimed_prior,
+    )
+

+def commit_state_change(new_state_obj: dict, state_path: Path | str, event: dict, *, root: str | None = None) -> str:
+    state_path = Path(state_path)
+    tmp = state_path.parent / f"{state_path.name}.tmp-{os.getpid()}"
+    content = json.dumps(new_state_obj, indent=2, ensure_ascii=False)

+        event_id = emit_event(**event, root=root)
+    except EventCommitError:
+        try:
+            os.unlink(str(tmp))
+        except OSError:
+            pass
+        raise
+

+def assert_lease_held(task: dict, holder: str, session: str, *, now: datetime | None = None, skew_minutes: int = 5) -> None:
+    resolved_now = _now(now)
+    lease = task.get("lease") if isinstance(task, dict) else None
+    task_id = task.get("id") if isinstance(task, dict) else "?"
+    if not lease:
+        raise LeaseLostError(f"no lease present on task {task_id}")
+    if lease.get("holder") != holder or lease.get("session") != session:
+        raise LeaseLostError(
+            f"task {task_id}: lease held by holder={lease.get('holder')!r} "
+            f"session={lease.get('session')!r}, not {holder!r}/{session!r}"
+        )
+    expires = _parse_iso_dt(lease.get("expires"))  # CorruptStateError propagates if unparseable
+    if resolved_now > expires + timedelta(minutes=skew_minutes):
+        raise LeaseLostError(
+            f"task {task_id}: lease expired at {expires.isoformat()} (+{skew_minutes}m grace)"
+        )

+# 1.10 Event emission — ADR-0004 discipline
+# ---------------------------------------------------------------------------

+_EVENT_SEQ = itertools.count(1)

+
+def _atomic_append_ndjson(log_path: Path, line: str) -> None:
+    """mtime-guarded read-all -> append line -> os.replace. Raises on
+    mtime-changed-mid-read or any I/O error; caller decides retry/spool."""
+    log_path = Path(log_path)
+    os.makedirs(log_path.parent, exist_ok=True)
+    if log_path.exists():
+        st_before = os.stat(log_path).st_mtime_ns
+        with open(log_path, "r", encoding="utf-8") as fh:
+            content = fh.read()
+        if os.stat(log_path).st_mtime_ns != st_before:
+            raise RuntimeError(f"event log changed mid-read (mtime guard tripped): {log_path}")
+    else:
+        content = ""
+    if content and not content.endswith("\n"):
+        content += "\n"
+    new_content = content + line + "\n"
+
+    tmp = log_path.parent / f"{log_path.name}.tmp-{os.getpid()}"
+    atomic_write(tmp, new_content)
+    os.replace(str(tmp), str(log_path))

+def spool_event(event: dict, project_id, *, root: str | None = None) -> Path:
+    spool_dir = event_spool_dir(project_id, root=root)
+    os.makedirs(spool_dir, exist_ok=True)
+    target = spool_dir / f"{time.time_ns()}-{os.getpid()}.json"
+    atomic_write(target, json.dumps(event, ensure_ascii=False))
+    return target

+def drain_event_spool(project_id, *, root: str | None = None, attempts: int = 3) -> int:
+    """Replay spooled events oldest-first into the NDJSON. Stops at the
+    first one that still can't be appended (leaves it + the rest for next
+    time). A malformed spooled file is discarded so it can't wedge the
+    queue forever. Returns the count successfully drained."""
+    spool_dir = event_spool_dir(project_id, root=root)
+    if not spool_dir.is_dir():
+        return 0
+
+    log_path = events_log_path(project_id, root=root)
+    names = sorted(p.name for p in spool_dir.iterdir() if p.is_file() and p.name.endswith(".json"))
+
+    drained = 0
+    for name in names:
+        p = spool_dir / name
+        try:
+            event = json.loads(p.read_text(encoding="utf-8"))
+        except (OSError, json.JSONDecodeError):
+            try:
+                p.unlink()
+            except OSError:
+                pass
+            continue
+
+        line = json.dumps(event, ensure_ascii=False)
+        done = False
+        for _attempt in range(attempts):
+            try:
+                _atomic_append_ndjson(log_path, line)
+                done = True
+                break
+            except Exception:
+                continue
+        if not done:
+            break  # still locked -- retry this + the rest on a later run
+
+        try:
+            p.unlink()
+        except OSError:
+            pass
+        drained += 1
+
+    return drained
+

+def emit_event(
+    type: str, actor: str, project_id: str,
+    refs: dict | None = None, detail: str | None = None,
+    event_log_ref: str | None = None, *,
+    now: datetime | None = None, root: str | None = None,
+) -> str:
+    resolved_now = _now(now)
+
+    # 1. Drain first.
+    drain_event_spool(project_id, root=root)
+
+    # 2. Build & validate.
+    event = {
+        "protocol_version": "1.0",
+        "event_id": f"e-{resolved_now.isoformat()}-{next(_EVENT_SEQ)}",
+        "type": type,
+        "actor": actor,
+        "timestamp": resolved_now.isoformat(),
+        "project_id": project_id,
+        "refs": refs or {},
+    }
+    if detail is not None:
+        event["detail"] = detail
+    if event_log_ref is not None:
+        event["event_log_ref"] = event_log_ref
+
+    errors = validate_obj(event, "event")
+    if errors:
+        raise EventCommitError(f"event failed schema validation, not appended: {errors}")
+
+    log_path = events_log_path(project_id, root=root)
+    line = json.dumps(event, ensure_ascii=False)
+
+    # 3. Append with <=1 retry, then spool.
+    appended = False
+    last_exc: Exception | None = None
+    for _attempt in range(2):
+        try:
+            _atomic_append_ndjson(log_path, line)
+            appended = True
+            break
+        except Exception as exc:
+            last_exc = exc
+
+    if not appended:
+        try:
+            spool_event(event, project_id, root=root)
+        except Exception as spool_exc:
+            # 4. EventCommitError only if both append and spool fail.
+            raise EventCommitError(
+                f"failed to append ({last_exc}) and failed to spool ({spool_exc})"
+            ) from spool_exc
+
+    return event["event_id"]

+#!/usr/bin/env python3
+"""
+tools/derive-heartbeat.py — AAOS Phase 3 §4.1: derive heartbeat.json from an
+externally-fetched signals payload.
+
+This is a pure local-file compute script. It never calls Notion or any MCP
+tool — per docs/PHASE3-DESIGN.md §4 and §6, and the zero-new-emitters
+doctrine in docs/adr/0005-heartbeat-derivation.md, the actual Notion /
+fleet-status.py fetch that produces the `signals.json` input happens at a
+future agent-session call site, out of Phase-3 scope. This script's only
+job is: read that already-fetched signal data, apply the freshness-banding
+math, cross-reference the agent registry, validate, and write
+heartbeat.json atomically.
+
+CLI:
+    python tools/derive-heartbeat.py --signals <path|-> [--out <path>]
+                                      [--now <iso>] [--seam-root <dir>]
+
+Input shape (docs/PHASE3-DESIGN.md §4.1):
+    {"agents": [
+        {"agent_id": "cowork", "last_run": "2026-07-09T06:00:00-04:00",
+         "source": ["notion:35c6ecd8 last_edited_time", "..."],
+         "threshold_hours": 16},
+        ...
+    ]}
+
+Status banding (delta = now - last_run, in hours):
+    last_run null/missing  -> "unknown"          (fail-closed)
+    delta <= threshold_hours       -> "fresh"
+    delta <= 2 * threshold_hours   -> "stale"
+    delta >  2 * threshold_hours   -> "missed"
+`source` and `threshold_hours` are always carried through from the input
+verbatim — inherited, never invented.
+
+Registry cross-reference (via _aaos_common.load_registry(), which always
+reads the git-tree REPO_ROOT/agent-registry.json — see design §0.2; there
+is deliberately no --registry override lever here, matching load_registry's
+own signature):
+  - A registered agent absent from signals.json is included with
+    status="unknown", last_run=null, source carried from that agent's own
+    `heartbeat_source` registry field (never invented), and
+    threshold_hours=0 (see the design-choice note below — there is no
+    threshold to inherit for an agent we observed nothing about, and the
+    schema requires the field regardless; "unknown" status makes the
+    numeric value moot to every consumer per the heartbeat schema's own
+    documented consumer note).
+  - A registered agent with `"active": false` is always forced to
+    status="unknown", regardless of what signals.json says about it (an
+    inactive agent is not part of the live fleet expire-leases.py should
+    reason about). Its last_run/source/threshold_hours are still carried
+    through from the observed signal when one exists (audit trail, not
+    fabrication) or from the registry/default-0 fallback when it doesn't.
+  - A signals.json entry for an agent_id that is NOT in the registry at all
+    is still surfaced (computed normally) rather than silently dropped —
+    refusing to report a real observed signal would itself be a kind of
+    fabrication-by-omission. The registry is the *expected* set (per the
+    "registry agent absent from signals" rule above), not an exhaustive
+    filter on what may appear.
+
+Exit codes:
+    0   success (heartbeat.json written; the printed line is its path)
+    1   assembled document failed schema validation — nothing written
+    2   usage error: --signals unreadable/missing, or its content is not
+        well-formed per the shape above (never fabricate a value for a
+        malformed field — refuse instead)
+"""
+from __future__ import annotations
+
+import argparse
+import json
+import sys
+from datetime import datetime
+from pathlib import Path
+
+SCRIPT_DIR = Path(__file__).resolve().parent
+if str(SCRIPT_DIR) not in sys.path:
+    sys.path.insert(0, str(SCRIPT_DIR))
+
+import _aaos_common as aaos  # noqa: E402  (import after sys.path manipulation, by design)
+
+# threshold_hours representation for a registry agent that has no signal at
+# all to inherit from (see module docstring's "Registry cross-reference").
+NO_SIGNAL_THRESHOLD_HOURS = 0
+
+
+def _read_signals_text(signals_arg: str) -> str:
+    if signals_arg == "-":
+        return sys.stdin.read()
+    path = Path(signals_arg)
+    try:
+        return path.read_text(encoding="utf-8")
+    except OSError as exc:
+        raise aaos.UsageError(f"could not read --signals file '{path}': {exc}") from exc
+
+
+def _parse_signals(raw_text: str) -> dict:
+    """Parse and validate the signals payload shape. Returns a dict keyed by
+    agent_id -> {"agent_id", "last_run_raw", "last_run_dt", "source",
+    "threshold_hours"}, preserving input order. Raises UsageError on any
+    malformed field — this function never fabricates a value."""
+    try:
+        data = json.loads(raw_text)
+    except json.JSONDecodeError as exc:
+        raise aaos.UsageError(f"--signals content is not valid JSON: {exc}") from exc
+
+    if not isinstance(data, dict) or not isinstance(data.get("agents"), list):
+        raise aaos.UsageError(
+            "--signals content must be a JSON object with an 'agents' array"
+        )
+
+    by_id: dict[str, dict] = {}
+    for i, entry in enumerate(data["agents"]):
+        if not isinstance(entry, dict):
+            raise aaos.UsageError(f"--signals agents[{i}] is not an object")
+
+        agent_id = entry.get("agent_id")
+        if not isinstance(agent_id, str) or not agent_id:
+            raise aaos.UsageError(f"--signals agents[{i}] missing a valid 'agent_id'")

+        if not isinstance(source, list) or not source or not all(
+            isinstance(s, str) for s in source

+                f"--signals agents[{i}] ('{agent_id}') 'source' must be a "
+                "non-empty array of strings"
+            )
+
+        threshold_hours = entry.get("threshold_hours")
+        if not isinstance(threshold_hours, (int, float)) or isinstance(threshold_hours, bool):
+            raise aaos.UsageError(
+                f"--signals agents[{i}] ('{agent_id}') 'threshold_hours' must be a number"
+            )

+        last_run_raw = entry.get("last_run", None)
+        last_run_dt = None
+        if last_run_raw is not None:
+            if not isinstance(last_run_raw, str):

+                    f"--signals agents[{i}] ('{agent_id}') 'last_run' must be a string or null"
+                )

+            except aaos.CorruptStateError as exc:
+                raise aaos.UsageError(
+                    f"--signals agents[{i}] ('{agent_id}') 'last_run' is unparseable: {exc}"
+                ) from exc

+            "agent_id": agent_id,
+            "last_run_raw": last_run_raw,
+            "last_run_dt": last_run_dt,
+            "source": list(source),
+            "threshold_hours": threshold_hours,
+        }
+    return by_id

+def _status_for(last_run_dt, threshold_hours, now: datetime) -> str:
+    if last_run_dt is None:
+        return "unknown"
+    delta_hours = (now - last_run_dt).total_seconds() / 3600.0
+    if delta_hours < 0:
+        # last_run reported in the future relative to now (clock skew) --
+        # never fabricate a worse status than the data actually supports.
+        return "fresh"
+    if delta_hours <= threshold_hours:
+        return "fresh"
+    if delta_hours <= 2 * threshold_hours:
+        return "stale"
+    return "missed"
+
+
+def _registry_heartbeat_source(reg_agent: dict, agent_id: str) -> list[str]:
+    source = reg_agent.get("heartbeat_source")
+    if isinstance(source, list) and source and all(isinstance(s, str) for s in source):
+        return source
+    # Defensive fallback only: the schema requires a non-empty `source`
+    # array, and a registry entry should always declare heartbeat_source,
+    # but refuse to leave the field unpopulated if it somehow doesn't.
+    return [f"agent-registry.json:{agent_id} (no heartbeat_source declared)"]
+
+
+def _build_heartbeat(signals_by_id: dict, registry: dict, now: datetime) -> dict:
+    agents_out: list[dict] = []
+    seen: set[str] = set()
+
+    for reg_agent in registry.get("agents", []):
+        if not isinstance(reg_agent, dict):
+            continue
+        agent_id = reg_agent.get("id")
+        if not isinstance(agent_id, str) or not agent_id:
+            continue
+        seen.add(agent_id)
+        active = reg_agent.get("active", True)
+        sig = signals_by_id.get(agent_id)
+
+        if sig is not None:
+            entry = {
+                "agent_id": agent_id,
+                "last_run": sig["last_run_raw"],
+                "source": sig["source"],
+                "status": _status_for(sig["last_run_dt"], sig["threshold_hours"], now),
+                "threshold_hours": sig["threshold_hours"],
+            }
+        else:
+            entry = {
+                "agent_id": agent_id,
+                "last_run": None,
+                "source": _registry_heartbeat_source(reg_agent, agent_id),
+                "status": "unknown",
+                "threshold_hours": NO_SIGNAL_THRESHOLD_HOURS,
+            }
+
+        if not active:
+            # Design §4.1 / task-instructions interpretation: an inactive
+            # registry agent is reported unknown regardless of what signals
+            # say -- it is not part of the live fleet expire-leases.py
+            # should reason about. Observed last_run/source/threshold_hours
+            # are still carried through verbatim above for audit purposes;
+            # only `status` is forced here.
+            entry["status"] = "unknown"
+
+        agents_out.append(entry)
+
+    for agent_id, sig in signals_by_id.items():
+        if agent_id in seen:
+            continue
+        agents_out.append({
+            "agent_id": agent_id,
+            "last_run": sig["last_run_raw"],
+            "source": sig["source"],
+            "status": _status_for(sig["last_run_dt"], sig["threshold_hours"], now),
+            "threshold_hours": sig["threshold_hours"],
+        })
+
+    return {
+        "protocol_version": "1.0",
+        "derived_at": now.isoformat(),
+        "agents": agents_out,
+    }
+
+
+def build_arg_parser() -> argparse.ArgumentParser:
+    parser = argparse.ArgumentParser(
+        prog="derive-heartbeat.py",
+        description="Derive heartbeat.json from an externally-fetched signals payload "
+        "(pure local-file compute -- never calls Notion/MCP itself).",
+    )
+    parser.add_argument(
+        "--signals", required=True,
+        help="Path to the signals JSON file, or '-' to read it from stdin.",
+    )
+    parser.add_argument(
+        "--out", default=None,

+    )

+        "--now", default=None,
+        help="ISO-8601 timestamp to use as 'now' (test hook; default: real clock).",
+    )

+        "--seam-root", default=None,
+        help="Override the seam root directory (default: $AAOS_SEAM_ROOT or the "
+        "built-in default).",
+    )
+    return parser
+
+

+    args = build_arg_parser().parse_args(argv)
+

+        try:
+            now = aaos._parse_iso_dt(args.now)
+        except aaos.CorruptStateError as exc:
+            print(f"USAGE ERROR: invalid --now value: {exc}")
